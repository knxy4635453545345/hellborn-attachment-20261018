-- ============================================================
-- MM2 HALLOWEEN FARM + FLING (lobby + round)
-- Speed: 20 | Limite: 50 | Velocity flight + coleta agressiva
-- ============================================================
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local LocalPlayer = Players.LocalPlayer

-- ================== CONFIG ==================
local CONFIG = {
    Speed = 20,
    MaxCoins = 50,
    CheckInterval = 1.0,
    AutoFarmEnabled = false,
    FlingPower = 1e5,
    FlingDuration = 0.6,
    UnderY = 4,
}

-- ================== ESTADO ==================
local isRunning = false
local runID = 0
local blacklist = {}
local BLACKLIST_TIME = 15
local log = {}
local function dbg(msg)
    local t = string.format("[%s] %s", os.date("%H:%M:%S"), tostring(msg))
    table.insert(log, 1, t)
    if #log > 6 then table.remove(log) end
    print(t)
end

-- ================== HELPERS ==================
local function getHRP()
    local c = LocalPlayer.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end
local function getHum()
    local c = LocalPlayer.Character
    return c and c:FindFirstChildOfClass("Humanoid")
end
local function shouldStop(myID)
    return not CONFIG.AutoFarmEnabled or myID ~= runID
end

local function getCoinCount()
    pcall(function()
        local c = LocalPlayer:FindFirstChild("Coins") or LocalPlayer:FindFirstChild("coins")
        if c and typeof(c) == "ValueBase" then return c.Value end
        local ls = LocalPlayer:FindFirstChild("leaderstats")
        if ls then
            local cc = ls:FindFirstChild("Coins") or ls:FindFirstChild("coins") or ls:FindFirstChild("Candy")
            if cc then return cc.Value end
        end
    end)
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if bp then
        local n = 0
        for _, i in ipairs(bp:GetChildren()) do
            if i.Name:lower():find("coin") or i.Name:lower():find("candy") then n = n + 1 end
        end
        return n
    end
    return 0
end
local function isInventoryFull() return getCoinCount() >= CONFIG.MaxCoins end

local function getPos(obj)
    if not obj or not obj.Parent then return nil end
    if obj:IsA("BasePart") then return obj.Position end
    if obj:IsA("Model") then
        if obj.PrimaryPart then return obj.PrimaryPart.Position end
        local p = obj:FindFirstChildWhichIsA("BasePart")
        return p and p.Position or nil
    end
    if obj:IsA("Tool") then
        local h = obj:FindFirstChild("Handle")
        return h and h.Position or nil
    end
    return nil
end

local function getTouchPart(obj)
    if not obj then return nil end
    if obj:IsA("BasePart") then return obj end
    if obj:IsA("Model") then return obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart") end
    if obj:IsA("Tool") then return obj:FindFirstChild("Handle") end
    return nil
end

local function isCoinName(name)
    name = name:lower()
    return name:find("coin") or name:find("candy") or name:find("treat")
        or name:find("sweet") or name:find("moeda") or name:find("doce")
end

local function isRealCoinOnMap(obj)
    if not obj or not obj.Parent then return false end
    if not (obj:IsA("BasePart") or obj:IsA("Model") or obj:IsA("Tool")) then return false end
    if not isCoinName(obj.Name) then return false end
    local cur = obj.Parent
    local inWorkspace = false
    while cur do
        if cur == workspace then inWorkspace = true; break end
        if cur:IsA("Player") then return false end
        if cur:IsA("Backpack") then return false end
        if cur:IsA("PlayerGui") then return false end
        if cur:IsA("Model") and cur:FindFirstChildOfClass("Humanoid") then return false end
        if cur == ReplicatedStorage then return false end
        if cur == game:GetService("ServerStorage") then return false end
        if cur == game:GetService("StarterPack") then return false end
        if cur == game:GetService("Lighting") then return false end
        cur = cur.Parent
    end
    if not inWorkspace then return false end
    local pos = getPos(obj)
    if not pos then return false end
    if pos.Y < -100 then return false end
    if pos.Magnitude < 1 then return false end
    if obj:IsA("BasePart") then
        if obj.Transparency >= 1 then return false end
    elseif obj:IsA("Model") then
        local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
        if part and part.Transparency >= 1 then return false end
    elseif obj:IsA("Tool") then
        local h = obj:FindFirstChild("Handle")
        if h and h.Transparency >= 1 then return false end
    end
    return true
end

local function getNearestCoin()
    local hrp = getHRP()
    if not hrp then return nil, nil end
    local nearest, shortest = nil, math.huge
    local now = tick()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("Model") or obj:IsA("Tool") then
            if isCoinName(obj.Name) then
                local bl = blacklist[obj]
                if not bl or (now - bl) > BLACKLIST_TIME then
                    if isRealCoinOnMap(obj) then
                        local pos = getPos(obj)
                        if pos then
                            local d = (pos - hrp.Position).Magnitude
                            if d < shortest then shortest = d; nearest = obj end
                        end
                    end
                end
            end
        end
    end
    return nearest, shortest
end

-- ================== DETECTAR LOBBY ==================
local function isInLobby()
    -- 1) Marcadores explícitos
    for _, n in ipairs({"Lobby", "WaitingRoom", "LobbyMap"}) do
        if workspace:FindFirstChild(n) then return true end
    end
    -- 2) Atributos de round
    for _, a in ipairs({"RoundActive", "InRound", "IsRoundActive", "RoundStarted"}) do
        local v = workspace:GetAttribute(a)
        if v == nil then v = ReplicatedStorage:GetAttribute(a) end
        if v ~= nil then return not v end
    end
    -- 3) Existe alguma moeda no mapa? Se não → provavelmente lobby
    for _, obj in ipairs(workspace:GetDescendants()) do
        if isCoinName(obj.Name) and isRealCoinOnMap(obj) then
            return false
        end
    end
    -- 4) Nenhum player tem papel ativo (Role)
    local anyRole = false
    for _, p in ipairs(Players:GetPlayers()) do
        local r = p:FindFirstChild("Role")
        if r and r:IsA("StringValue") and r.Value ~= "" then anyRole = true; break end
    end
    return not anyRole
end

-- ================== VOO POR VELOCIDADE (física real) ==================
local function flyTo(targetPos, speed, myID)
    local hrp = getHRP()
    if not hrp then return false end

    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.Velocity = Vector3.zero
    bv.Parent = hrp

    local function cleanup()
        pcall(function() bv:Destroy() end)
    end

    while true do
        task.wait(0.03)
        if shouldStop(myID) then cleanup(); return false end
        local cur = getHRP()
        if not cur or not cur.Parent then cleanup(); return false end

        local delta = targetPos - cur.Position
        local dist = delta.Magnitude
        if dist < 1 then break end

        local v = math.min(speed, dist * 8)
        bv.Velocity = delta.Unit * v
    end

    cleanup()
    return true
end

-- ================== COLETA AGRESSIVA ==================
local function forceCollect(obj)
    local hrp = getHRP()
    local part = getTouchPart(obj)
    if not hrp or not part then return end

    -- Garante colisão
    pcall(function() part.CanTouch = true end)
    pcall(function()
        if sethiddenproperty then sethiddenproperty(part, "CanTouch", true) end
    end)

    -- MÉTODO 1: firetouchinterest (mais confiável)
    if firetouchinterest then
        pcall(function()
            firetouchinterest(hrp, part, 0)
            firetouchinterest(part, hrp, 0)
        end)
        pcall(function()
            firetouchinterest(hrp, part, 1)
            firetouchinterest(part, hrp, 1)
        end)
    end

    -- MÉTODO 2: firesignal (executor com getconnections/firesignal)
    if firesignal then
        pcall(function() firesignal(part.Touched, hrp) end)
        pcall(function() firesignal(hrp.Touched, part) end)
        pcall(function() firesignal(part.TouchInterest) end)
    end

    -- MÉTODO 3: getconnections:Fire
    if getconnections then
        pcall(function()
            for _, c in ipairs(getconnections(part.Touched)) do
                pcall(function() c:Fire(hrp) end)
            end
        end)
        pcall(function()
            for _, c in ipairs(getconnections(hrp.Touched)) do
                pcall(function() c:Fire(part) end)
            end
        end)
    end

    -- MÉTODO 4: Touched:Fire (raramente funciona, mas tenta)
    pcall(function() part.Touched:Fire(hrp) end)
    pcall(function() hrp.Touched:Fire(part) end)

    -- MÉTODO 5: ClickDetector
    local cd = part:FindFirstChildOfClass("ClickDetector")
        or obj:FindFirstChildOfClass("ClickDetector")
    if cd and fireclickdetector then
        pcall(function() fireclickdetector(cd) end)
    end

    -- MÉTODO 6: ProximityPrompt
    local pp = part:FindFirstChildOfClass("ProximityPrompt")
        or obj:FindFirstChildOfClass("ProximityPrompt")
    if pp and fireproximityprompt then
        pcall(function() fireproximityprompt(pp) end)
    end
end

local function collectCoin(coin, myID)
    local coinPos = getPos(coin)
    if not coinPos then return false end

    -- FASE 1: ir para baixo da moeda (física via velocity)
    if not flyTo(coinPos - Vector3.new(0, CONFIG.UnderY, 0), CONFIG.Speed, myID) then return false end
    if shouldStop(myID) then return false end
    if not coin.Parent then return true end

    -- FASE 2: subir atravessando com velocity (Touched dispara no servidor)
    local hrp = getHRP()
    if not hrp then return false end
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.Parent = hrp

    local steps = 12
    for i = 0, steps do
        if shouldStop(myID) then break end
        if not coin.Parent then break end
        local cur = getHRP()
        if not cur then break end

        local y = -CONFIG.UnderY + (i / steps) * (CONFIG.UnderY * 2)
        local target = coinPos + Vector3.new(0, y, 0)
        local delta = target - cur.Position
        bv.Velocity = delta.Unit * math.min(CONFIG.Speed, delta.Magnitude * 10)
        forceCollect(coin)
        task.wait(0.05)
    end

    pcall(function() bv:Destroy() end)

    -- FASE 3: jitter final com CFrame + todos métodos
    for i = 1, 6 do
        if shouldStop(myID) then break end
        if not coin.Parent then break end
        local cur = getHRP()
        if not cur then break end
        local jitter = Vector3.new(
            math.random(-8, 8) / 10,
            math.random(-5, 10) / 10,
            math.random(-8, 8) / 10
        )
        pcall(function() cur.CFrame = CFrame.new(coinPos + jitter) end)
        forceCollect(coin)
        task.wait(0.05)
    end

    return not coin.Parent or not isRealCoinOnMap(coin)
end

-- ================== FLING ==================
local function flingTarget(targetChar)
    if not targetChar then return false end
    local hrp = targetChar:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    local hum = targetChar:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end

    pcall(function() hrp:SetNetworkOwner(LocalPlayer) end)
    pcall(function() hrp:SetNetworkOwnershipAuto() end)

    local bv = Instance.new("BodyVelocity")
    bv.Velocity = Vector3.new(CONFIG.FlingPower, CONFIG.FlingPower, CONFIG.FlingPower)
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.Parent = hrp

    local bav = Instance.new("BodyAngularVelocity")
    bav.AngularVelocity = Vector3.new(CONFIG.FlingPower, CONFIG.FlingPower, CONFIG.FlingPower)
    bav.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    bav.Parent = hrp

    local conn = RunService.Heartbeat:Connect(function()
        pcall(function()
            if hrp.Parent then
                hrp.AssemblyLinearVelocity = Vector3.new(CONFIG.FlingPower, CONFIG.FlingPower, CONFIG.FlingPower)
                hrp.AssemblyAngularVelocity = Vector3.new(CONFIG.FlingPower, CONFIG.FlingPower, CONFIG.FlingPower)
            end
        end)
    end)

    task.wait(CONFIG.FlingDuration)
    conn:Disconnect()
    pcall(function() bv:Destroy() end)
    pcall(function() bav:Destroy() end)
    dbg("FLING → " .. (targetChar.Name or "?"))
    return true
end

-- ================== DETECTAR MURDER ==================
local function isPlayerMurderer(player)
    if not player or player == LocalPlayer then return false end
    local char = player.Character
    if not char then return false end
    local function check(c)
        if not c then return false end
        for _, t in ipairs(c:GetChildren()) do
            if t:IsA("Tool") and t.Name:lower():find("knife") then return true end
        end
        return false
    end
    if check(player:FindFirstChild("Backpack")) then return true end
    if check(char) then return true end
    local r = player:FindFirstChild("Role")
    if r and r:IsA("StringValue") and r.Value:lower() == "murderer" then return true end
    return false
end
local function findMurderer()
    for _, p in ipairs(Players:GetPlayers()) do
        if isPlayerMurderer(p) then return p end
    end
    return nil
end

-- ================== PAPEL ==================
local function getRole()
    local r = LocalPlayer:FindFirstChild("Role") or workspace:FindFirstChild(LocalPlayer.Name .. "_Role")
    if r and r:IsA("StringValue") then return r.Value:lower() end
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if bp then
        for _, t in ipairs(bp:GetChildren()) do
            if t.Name:lower():find("knife") then return "murderer" end
            if t.Name:lower():find("gun") or t.Name:lower():find("revolver") then return "sheriff" end
        end
    end
    return "innocent"
end

-- ================== FLING NO MURDER (lobby ou inventário cheio) ==================
local function attackMurder(myID)
    local role = getRole()
    if role == "murderer" then return end -- assassino não flinga a si mesmo

    local murderer = findMurderer()
    if not murderer then
        dbg("Murder não encontrado")
        return
    end
    local mchar = murderer.Character
    local mhrp = mchar and mchar:FindFirstChild("HumanoidRootPart")
    if not mhrp then return end

    dbg("Indo FLINGAR " .. murderer.Name)
    if not flyTo(mhrp.Position, CONFIG.Speed, myID) then return end
    if shouldStop(myID) then return end
    task.wait(0.1)

    if role == "sheriff" then
        local gun = LocalPlayer.Backpack and LocalPlayer.Backpack:FindFirstChildOfClass("Tool")
        if gun then pcall(function() gun:Activate() end); task.wait(0.1) end
    end

    flingTarget(mchar)

    for i = 1, 3 do
        task.wait(0.5)
        if shouldStop(myID) then return end
        if not mchar or not mchar.Parent then break end
        local h = mchar:FindFirstChildOfClass("Humanoid")
        if h and h.Health <= 0 then break end
        local th = mchar:FindFirstChild("HumanoidRootPart")
        if not th then break end
        if not flyTo(th.Position, CONFIG.Speed, myID) then return end
        flingTarget(mchar)
    end
end

-- ================== LOOP PRINCIPAL ==================
local function stepFarm(myID)
    if shouldStop(myID) then return end
    if not getHRP() then task.wait(0.5); return end

    -- 🏠 LOBBY → sempre fling murder (não precisa de moedas cheias)
    if isInLobby() then
        local murderer = findMurderer()
        if murderer then
            dbg("Lobby detectado → fling no murder")
            attackMurder(myID)
            task.wait(1.5)
        else
            dbg("Lobby, mas murder não encontrado")
            task.wait(1.5)
        end
        return
    end

    -- 🎮 ROUND → inventário cheio? fling murder
    if isInventoryFull() then
        attackMurder(myID)
        task.wait(CONFIG.CheckInterval)
        return
    end

    -- 🪙 ROUND → coletar moedas
    local coin, dist = getNearestCoin()
    if coin then
        dbg(string.format("Coletando moeda (%.0f studs)", dist or 0))
        local ok = collectCoin(coin, myID)
        if shouldStop(myID) then return end
        if ok then
            dbg("Moeda coletada ✔")
        else
            blacklist[coin] = tick()
            dbg("Moeda não coletou → blacklist")
        end
        task.wait(0.05)
    else
        dbg("Nenhuma moeda no mapa, aguardando...")
        task.wait(0.8)
    end
end

local function startLoop()
    runID = runID + 1
    local myID = runID
    isRunning = true
    task.spawn(function()
        dbg("Farm iniciado (gen " .. myID .. ")")
        while not shouldStop(myID) do
            local ok, err = pcall(stepFarm, myID)
            if not ok then
                dbg("Erro (recuperado): " .. tostring(err))
                task.wait(0.5)
            end
            RunService.Heartbeat:Wait()
        end
        if myID == runID then
            isRunning = false
            dbg("Farm parado (gen " .. myID .. ")")
        end
    end)
end

local function stopLoop()
    CONFIG.AutoFarmEnabled = false
    runID = runID + 1
    isRunning = false
    local hrp = getHRP()
    if hrp and hrp.Parent then hrp.Anchored = false end
    local hum = getHum()
    if hum then hum.WalkSpeed = 16 end
    dbg("STOP forçado")
end

-- ================== WATCHDOG ==================
task.spawn(function()
    while true do
        task.wait(1)
        pcall(function()
            if not CONFIG.AutoFarmEnabled then
                local hrp = getHRP()
                if hrp and hrp.Anchored then hrp.Anchored = false end
            end
            if CONFIG.AutoFarmEnabled and not isRunning then
                dbg("Watchdog: reiniciando")
                startLoop()
            end
        end)
    end
end)

    task.wait(0.5)
