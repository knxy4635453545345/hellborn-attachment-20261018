-- ============================================================
-- MM2 HALLOWEEN AUTO FARM (VOANDO) + FLING NO MURDER
-- Limite: 50 moedas | Speed: 20 | Respawn-safe | STOP instantâneo
-- ============================================================
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local LocalPlayer = Players.LocalPlayer

-- ================== CONFIGURAÇÕES ==================
local CONFIG = {
    Speed = 20,
    MaxCoins = 50,
    CheckInterval = 1.0,
    AutoFarmEnabled = false,
    FlingPower = 1e5,
    FlingDuration = 0.6,
}

-- ================== ESTADO ==================
local isRunning = false
local runID = 0                -- ⚡ invalida loops antigos
local blacklist = {}
local BLACKLIST_TIME = 15
local log = {}
local function dbg(msg)
    local t = string.format("[%s] %s", os.date("%H:%M:%S"), tostring(msg))
    table.insert(log, 1, t)
    if #log > 6 then table.remove(log) end
    print(t)
end

-- ================== HELPERS ==================
local function getHRP()
    local c = LocalPlayer.Character
    if not c then return nil end
    return c:FindFirstChild("HumanoidRootPart")
end
local function getHum()
    local c = LocalPlayer.Character
    if not c then return nil end
    return c:FindFirstChildOfClass("Humanoid")
end

-- ⚡ cancela se parar OU se um novo INICIAR aconteceu
local function shouldStop(myID)
    return not CONFIG.AutoFarmEnabled or myID ~= runID
end

local function getCoinCount()
    pcall(function()
        local c = LocalPlayer:FindFirstChild("Coins") or LocalPlayer:FindFirstChild("coins")
        if c and typeof(c) == "ValueBase" then return c.Value end
        local ls = LocalPlayer:FindFirstChild("leaderstats")
        if ls then
            local cc = ls:FindFirstChild("Coins") or ls:FindFirstChild("coins") or ls:FindFirstChild("Candy")
            if cc then return cc.Value end
        end
    end)
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if bp then
        local n = 0
        for _, i in ipairs(bp:GetChildren()) do
            if i.Name:lower():find("coin") or i.Name:lower():find("candy") then n = n + 1 end
        end
        return n
    end
    return 0
end
local function isInventoryFull() return getCoinCount() >= CONFIG.MaxCoins end

local function getPos(obj)
    if not obj or not obj.Parent then return nil end
    if obj:IsA("BasePart") then return obj.Position end
    if obj:IsA("Model") then
        if obj.PrimaryPart then return obj.PrimaryPart.Position end
        local p = obj:FindFirstChildWhichIsA("BasePart")
        return p and p.Position or nil
    end
    if obj:IsA("Tool") then
        local h = obj:FindFirstChild("Handle")
        return h and h.Position or nil
    end
    return nil
end

local function isCoinName(name)
    name = name:lower()
    return name:find("coin") or name:find("candy") or name:find("treat")
        or name:find("sweet") or name:find("moeda") or name:find("doce")
end

local function isRealCoinOnMap(obj)
    if not obj or not obj.Parent then return false end
    if not (obj:IsA("BasePart") or obj:IsA("Model") or obj:IsA("Tool")) then return false end
    if not isCoinName(obj.Name) then return false end
    local cur = obj.Parent
    local inWorkspace = false
    while cur do
        if cur == workspace then inWorkspace = true; break end
        if cur:IsA("Player") then return false end
        if cur:IsA("Backpack") then return false end
        if cur:IsA("PlayerGui") then return false end
        if cur:IsA("Model") and cur:FindFirstChildOfClass("Humanoid") then return false end
        if cur == game:GetService("ReplicatedStorage") then return false end
        if cur == game:GetService("ServerStorage") then return false end
        if cur == game:GetService("StarterPack") then return false end
        if cur == game:GetService("Lighting") then return false end
        cur = cur.Parent
    end
    if not inWorkspace then return false end
    local pos = getPos(obj)
    if not pos then return false end
    if pos.Y < -100 then return false end
    if pos.Magnitude < 1 then return false end
    if obj:IsA("BasePart") then
        if obj.Transparency >= 1 then return false end
    elseif obj:IsA("Model") then
        local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
        if part and part.Transparency >= 1 then return false end
    elseif obj:IsA("Tool") then
        local h = obj:FindFirstChild("Handle")
        if h and h.Transparency >= 1 then return false end
    end
    return true
end

local function getNearestCoin()
    local hrp = getHRP()
    if not hrp then return nil, nil end
    local nearest, shortest = nil, math.huge
    local now = tick()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("Model") or obj:IsA("Tool") then
            if isCoinName(obj.Name) then
                local bl = blacklist[obj]
                if not bl or (now - bl) > BLACKLIST_TIME then
                    if isRealCoinOnMap(obj) then
                        local pos = getPos(obj)
                        if pos then
                            local d = (pos - hrp.Position).Magnitude
                            if d < shortest then shortest = d; nearest = obj end
                        end
                    end
                end
            end
        end
    end
    return nearest, shortest
end

-- ================== VOO (cancela se parar) ==================
local function flyTo(targetPos, speed, myID)
    local hrp = getHRP()
    local hum = getHum()
    if not hrp or not hum then return false end

    local startPos = hrp.Position
    local dist = (targetPos - startPos).Magnitude
    if dist < 1 then return true end

    local oldWS = hum.WalkSpeed
    hrp.Anchored = true
    hum.WalkSpeed = 0

    local dur = dist / math.max(speed, 1)
    local t0 = tick()

    while true do
        task.wait(0.03)
        if shouldStop(myID) then
            -- ⚡ parada imediata
            local cur = getHRP()
            if cur and cur.Parent then cur.Anchored = false end
            local h = getHum()
            if h then h.WalkSpeed = oldWS end
            return false
        end
        local cur = getHRP()
        if not cur or not cur.Parent then break end
        local a = math.min((tick() - t0) / dur, 1)
        cur.CFrame = CFrame.new(startPos:Lerp(targetPos, a))
        if a >= 1 then break end
    end

    local finalHRP = getHRP()
    local finalHum = getHum()
    if finalHRP and finalHRP.Parent then finalHRP.Anchored = false end
    if finalHum then finalHum.WalkSpeed = oldWS end
    return not shouldStop(myID)
end

-- ================== FLING ==================
local function flingTarget(targetChar)
    if not targetChar then return false end
    local hrp = targetChar:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    local hum = targetChar:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end

    pcall(function() hrp:SetNetworkOwner(LocalPlayer) end)
    pcall(function() hrp:SetNetworkOwnershipAuto() end)

    local bv = Instance.new("BodyVelocity")
    bv.Velocity = Vector3.new(CONFIG.FlingPower, CONFIG.FlingPower, CONFIG.FlingPower)
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.Parent = hrp

    local bav = Instance.new("BodyAngularVelocity")
    bav.AngularVelocity = Vector3.new(CONFIG.FlingPower, CONFIG.FlingPower, CONFIG.FlingPower)
    bav.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    bav.Parent = hrp

    local conn = RunService.Heartbeat:Connect(function()
        pcall(function()
            if hrp.Parent then
                hrp.AssemblyLinearVelocity = Vector3.new(CONFIG.FlingPower, CONFIG.FlingPower, CONFIG.FlingPower)
                hrp.AssemblyAngularVelocity = Vector3.new(CONFIG.FlingPower, CONFIG.FlingPower, CONFIG.FlingPower)
            end
        end)
    end)

    task.wait(CONFIG.FlingDuration)
    conn:Disconnect()

    pcall(function() bv:Destroy() end)
    pcall(function() bav:Destroy() end)
    dbg("FLING aplicado em " .. (targetChar.Name or "?"))
    return true
end

-- ================== DETECTAR MURDER ==================
local function isPlayerMurderer(player)
    if not player or player == LocalPlayer then return false end
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end

    local function checkForKnife(container)
        if not container then return false end
        for _, t in ipairs(container:GetChildren()) do
            if t:IsA("Tool") and t.Name:lower():find("knife") then return true end
        end
        return false
    end

    if checkForKnife(player:FindFirstChild("Backpack")) then return true end
    if checkForKnife(char) then return true end

    local r = player:FindFirstChild("Role")
    if r and r:IsA("StringValue") and r.Value:lower() == "murderer" then return true end

    return false
end

local function findMurderer()
    for _, p in ipairs(Players:GetPlayers()) do
        if isPlayerMurderer(p) then return p end
    end
    return nil
end

-- ================== PAPEL ==================
local function getRole()
    local r = LocalPlayer:FindFirstChild("Role") or workspace:FindFirstChild(LocalPlayer.Name .. "_Role")
    if r and r:IsA("StringValue") then return r.Value:lower() end
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if bp then
        for _, t in ipairs(bp:GetChildren()) do
            if t.Name:lower():find("knife") then return "murderer" end
            if t.Name:lower():find("gun") or t.Name:lower():find("revolver") then return "sheriff" end
        end
    end
    return "innocent"
end

-- ================== AÇÃO AO ENCHER INVENTÁRIO (com check de cancelamento) ==================
local function executeRoleAction(myID)
    local role = getRole()

    if role == "murderer" then
        dbg("Assassino cheio → matando todos")
        for _, p in ipairs(Players:GetPlayers()) do
            if shouldStop(myID) then return end
            if p ~= LocalPlayer and p.Character then
                local th = p.Character:FindFirstChild("HumanoidRootPart")
                if th then
                    if not flyTo(th.Position, CONFIG.Speed, myID) then return end
                    local k = LocalPlayer.Backpack and LocalPlayer.Backpack:FindFirstChildOfClass("Tool")
                    if k then k:Activate() end
                    task.wait(0.4)
                end
            end
        end
        return
    end

    local murderer = findMurderer()
    if not murderer then
        dbg("Inventário cheio, mas murder não encontrado → aguardando")
        return
    end
    local mchar = murderer.Character
    local mhrp = mchar and mchar:FindFirstChild("HumanoidRootPart")
    if not mhrp then return end

    dbg("Inventário cheio → indo FLINGAR " .. murderer.Name)
    if not flyTo(mhrp.Position, CONFIG.Speed, myID) then return end
    task.wait(0.1)
    if shouldStop(myID) then return end

    if role == "sheriff" then
        local gun = LocalPlayer.Backpack and LocalPlayer.Backpack:FindFirstChildOfClass("Tool")
        if gun then
            pcall(function() gun:Activate() end)
            task.wait(0.1)
        end
    end

    flingTarget(mchar)

    for i = 1, 3 do
        task.wait(0.5)
        if shouldStop(myID) then return end
        if not mchar or not mchar.Parent then break end
        local h = mchar:FindFirstChildOfClass("Humanoid")
        if h and h.Health <= 0 then break end
        local th = mchar:FindFirstChild("HumanoidRootPart")
        if not th then break end
        if not flyTo(th.Position, CONFIG.Speed, myID) then return end
        flingTarget(mchar)
    end
end

-- ================== LOOP PRINCIPAL ==================
local function stepFarm(myID)
    if shouldStop(myID) then return end
    if not getHRP() then task.wait(0.5); return end

    if isInventoryFull() then
        executeRoleAction(myID)
        task.wait(CONFIG.CheckInterval)
        return
    end

    local coin, dist = getNearestCoin()
    if coin then
        local pos = getPos(coin)
        if pos then
            dbg(string.format("Voando até moeda (%.0f studs)", dist or 0))
            if not flyTo(pos, CONFIG.Speed, myID) then return end
            task.wait(0.1)
            if coin.Parent and isRealCoinOnMap(coin) then
                blacklist[coin] = tick()
                dbg("Moeda não coletada → blacklist")
            else
                dbg("Moeda coletada ✔")
            end
        end
    else
        dbg("Nenhuma moeda spawnada, aguardando...")
        task.wait(0.8)
    end
end

local function startLoop()
    -- sempre cria uma nova geração (invalida a antiga)
    runID = runID + 1
    local myID = runID
    isRunning = true
    task.spawn(function()
        dbg("Farm iniciado (gen " .. myID .. ")")
        while not shouldStop(myID) do
            local ok, err = pcall(stepFarm, myID)
            if not ok then
                dbg("Erro (recuperado): " .. tostring(err))
                task.wait(0.5)
            end
            RunService.Heartbeat:Wait()
        end
        if myID == runID then
            isRunning = false
            dbg("Farm parado (gen " .. myID .. ")")
        end
    end)
end

-- ⚡ STOP forçado
local function stopLoop()
    CONFIG.AutoFarmEnabled = false
    runID = runID + 1              -- invalida imediatamente o loop atual
    isRunning = false
    local hrp = getHRP()
    if hrp and hrp.Parent then hrp.Anchored = false end
    local hum = getHum()
    if hum then hum.WalkSpeed = 16 end
    dbg("STOP forçado")
end

-- ================== WATCHDOG (sempre atualizando) ==================
-- Loop infinito que nunca para: mantém estado limpo e recupera de erros
task.spawn(function()
    while true do
        task.wait(1)
        pcall(function()
            -- Se o personagem existe mas está ancorado sem o farm estar voando, desancora
            if not CONFIG.AutoFarmEnabled then
                local hrp = getHRP()
                if hrp and hrp.Anchored then
                    hrp.Anchored = false
                end
            end
            -- Se o player morreu/respawnou e o loop morreu, reinicia
            if CONFIG.AutoFarmEnabled and not isRunning then
                dbg("Watchdog: loop morto → reiniciando")
                startLoop()
            end
        end)
    end
end)

-- ================== RESPAWN ==================
LocalPlayer.CharacterAdded:Connect(function(char)
    dbg("Respawn detectado → aguardando HRP")
    char:WaitForChild("HumanoidRootPart", 5)
    task.wait(0.5)
    if CONFIG.AutoFarmEnabled and not isRunning then
        dbg("Reiniciando farm no respawn")
        startLoop()
    end
end)

LocalPlayer.CharacterRemoving:Connect(function()
    dbg("Personagem morrendo/saindo")
    local hrp = getHRP()
    if hrp and hrp.Parent then hrp.Anchored = false end
end)
