local function findAllSendRemotes()
    if M.cachedSendRemotes and #M.cachedSendRemotes > 0 then
        -- Verifica se ainda são válidos
        local valid = {}
        for _, r in ipairs(M.cachedSendRemotes) do
            if r and r.Parent then table.insert(valid, r) end
        end
        if #valid > 0 then return valid end
    end
    
    local found = {}
    local seen = {}
    local function add(r)
        if r and not seen[r] and (r:IsA("RemoteEvent") or r:IsA("RemoteFunction")) then
            seen[r] = true
            table.insert(found, r)
        end
    end
    
    -- 1. Caminhos exatos (mais prováveis no MM2)
    local paths = {
        M.TradeFolder and M.TradeFolder:FindFirstChild("Remotes") and M.TradeFolder.Remotes:FindFirstChild("SendTradeRequest"),
        M.TradeFolder and M.TradeFolder:FindFirstChild("Remotes") and M.TradeFolder.Remotes:FindFirstChild("RequestTrade"),
        M.TradeFolder and M.TradeFolder:FindFirstChild("Remotes") and M.TradeFolder.Remotes:FindFirstChild("SendRequest"),
        M.TradeFolder and M.TradeFolder:FindFirstChild("Remotes") and M.TradeFolder.Remotes:FindFirstChild("Request"),
        M.TradeFolder and M.TradeFolder:FindFirstChild("SendTradeRequest"),
        M.TradeFolder and M.TradeFolder:FindFirstChild("RequestTrade"),
        M.TradeFolder and M.TradeFolder:FindFirstChild("Request"),
        M.TradeFolder and M.TradeFolder:FindFirstChild("Send"),
        M.TradeFolder and M.TradeFolder:FindFirstChild("RequestPlayer"),
        M.TradeFolder and M.TradeFolder:FindFirstChild("TradeRequest"),
        RS:FindFirstChild("Remotes") and RS.Remotes:FindFirstChild("SendTradeRequest"),
        RS:FindFirstChild("Remotes") and RS.Remotes:FindFirstChild("RequestTrade"),
        RS:FindFirstChild("Remotes") and RS.Remotes:FindFirstChild("Request"),
        RS:FindFirstChild("SendTradeRequest"),
        RS:FindFirstChild("RequestTrade"),
        RS:FindFirstChild("TradeRequest"),
    }
    for _, r in ipairs(paths) do add(r) end
    
    -- 2. Busca direta por nome (não recursiva, mas em vários parents)
    local parents = {M.TradeFolder, RS}
    if M.TradeFolder then
        table.insert(parents, M.TradeFolder:FindFirstChild("Remotes"))
        table.insert(parents, M.TradeFolder:FindFirstChild("Events"))
        table.insert(parents, M.TradeFolder:FindFirstChild("RemoteEvents"))
    end
    table.insert(parents, RS:FindFirstChild("Remotes"))
    table.insert(parents, RS:FindFirstChild("RemoteEvents"))
    table.insert(parents, RS:FindFirstChild("Events"))
    table.insert(parents, RS:FindFirstChild("TradeRemotes"))
    
    local names = {
        "SendTradeRequest", "RequestTrade", "TradeRequest", "InviteTrade",
        "SendRequest", "OfferTrade", "SendTrade", "RequestTradeEvent",
        "TradeRequestEvent", "SendTradeInvite", "Invite", "Send",
    }
    for _, parent in ipairs(parents) do
        if parent then
            for _, n in ipairs(names) do
                add(parent:FindFirstChild(n))
            end
        end
    end
    
    -- 3. Busca recursiva por nomes (RS:FindFirstChild(name, true))
    for _, n in ipairs(names) do
        local ok, r = pcall(function() return RS:FindFirstChild(n, true) end)
        if ok and r then add(r) end
    end
    
    -- 4. Fuzzy: qualquer remote com "trade" + "request/send/invite/offer"
    local ok = pcall(function()
        for _, obj in ipairs(RS:GetDescendants()) do
            if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
                local n = string.lower(obj.Name)
                if n:find("trade") and (n:find("request") or n:find("send") or n:find("invite") or n:find("offer")) then
                    add(obj)
                end
            end
        end
    end)
    
    M.cachedSendRemotes = found
    
    print("========================================")
    print("[REQUEST] Remotes candidatos encontrados:", #found)
    for i, r in ipairs(found) do
        print(string.format("  [%d] %s (%s)", i, r:GetFullName(), r.ClassName))
    end
    print("========================================")
    
    return found
end

local function findRemote(cands)
    local locs = {M.TradeFolder, RS, RS:FindFirstChild("Remotes"), RS:FindFirstChild("RemoteEvents")}
    if M.TradeFolder then
        table.insert(locs, M.TradeFolder:FindFirstChild("Remotes"))
        table.insert(locs, M.TradeFolder:FindFirstChild("RemoteEvents"))
    end
    for _, n in ipairs(cands) do
        for _, loc in ipairs(locs) do
            if loc then
                local r = loc:FindFirstChild(n)
                if r and (r:IsA("RemoteEvent") or r:IsA("RemoteFunction")) then return r end
            end
        end
    end
end

local function clean(n)
    if not n then return "" end
    n = string.lower(n)
    n = n:gsub("%b()", ""):gsub("x%d+", ""):gsub("%d+x", ""):gsub("%d+", "")
    n = n:gsub("['’]", ""):gsub("[^%w%s]", ""):gsub("%s+", " ")
    return n:match("^%s*(.-)%s*$")
end

M.VAL_CACHE = {}
M.DEM_CACHE = {}
for k, v in pairs(M.ITEM_VALUES) do M.VAL_CACHE[clean(k)] = v end
for k, v in pairs(M.ITEM_DEMAND) do M.DEM_CACHE[clean(k)] = v end

local function getVal(n)
    if not n or n == "" then return 0 end
    local c = clean(n)
    if c == "" then return 0 end
    if M.VAL_CACHE[c] then return M.VAL_CACHE[c] end
    if c:find("chroma") then
        for k, v in pairs(M.VAL_CACHE) do
            if k:find("chroma") and (k:find(c:gsub("chroma ", "")) or c:find(k:gsub("chroma ", ""))) then
                return v
            end
        end
    end
    for k, v in pairs(M.VAL_CACHE) do
        if k == c or k:find(c, 1, true) or c:find(k, 1, true) then return v end
    end
    return 0
end

local function getDemand(n)
    if not n or n == "" then return 1 end
    local c = clean(n)
    if M.DEM_CACHE[c] then return M.DEM_CACHE[c] end
    for k, v in pairs(M.DEM_CACHE) do
        if k:find(c, 1, true) or c:find(k, 1, true) then return v end
    end
    return 1
end

local function getExact(n)
    local c = clean(n)
    for k in pairs(M.ITEM_VALUES) do if clean(k) == c then return k end end
    return n
end

local function parseStack(t)
    if not t then return "", 1 end
    local q = 1
    local s = t:match("^%s*(.-)%s*$") or t
    local m, qv = s:match("^(.-)%s*%(?x%s*(%d+)%)?$")
    if m and m ~= "" and qv then s = m; q = tonumber(qv) end
    local q2, m2 = s:match("^(%d+)%s*x%s*(.+)$")
    if q2 and m2 then q = tonumber(q2); s = m2 end
    return s:match("^%s*(.-)%s*$") or s, q
end

local function fmt(v)
    if v >= 1e9 then return string.format("%.1fB", v/1e9)
    elseif v >= 1e6 then return string.format("%.1fM", v/1e6)
    elseif v >= 1e3 then return string.format("%.1fK", v/1e3)
    else return tostring(math.floor(v)) end
end

local function isSafe()
    if not LP.Character or not LP.Character.Parent then return false end
    local h = LP.Character:FindFirstChildOfClass("Humanoid")
    return h and h.Health > 0
end

local function getInv()
    local out = {}
    local ok, d = pcall(function() return require(RS.Modules.ProfileData) end)
    if ok and d and d.Weapons and d.Weapons.Owned then
        for k, c in pairs(d.Weapons.Owned) do
            if c > 0 and getVal(k) > 0 then
                for _ = 1, c do table.insert(out, k) end
            end
        end
    end
    return out
end

local function visible(o, r)
    local c = o
    while c and c ~= r do
        if c:IsA("GuiObject") and not c.Visible then return false end
        c = c.Parent
    end
    return true
end

local function folderStats(f)
    if not isSafe() or not f or not f.Parent then return 0, 0 end
    local cnt, tot = 0, 0
    for _, o in ipairs(f:GetDescendants()) do
        if o:IsA("GuiObject") and o.Visible and visible(o, f) then
            local t = (o:IsA("TextLabel") or o:IsA("TextButton") or o:IsA("TextBox")) and o.Text or ""
            local bn, q = parseStack(t)
            local uv = getVal(bn)
            if uv == 0 then bn, q = parseStack(o.Name); uv = getVal(bn) end
            if uv > 0 then cnt = cnt + q; tot = tot + (uv * q) end
        end
    end
    return cnt, tot
end

local function folderFull(f)
    if not isSafe() or not f or not f.Parent then return 0, 1, "—", 0 end
    local tv, qty, ds, mn, mv = 0, 0, 0, "—", 0
    for _, o in ipairs(f:GetDescendants()) do
        if o:IsA("GuiObject") and o.Visible and visible(o, f) then
            local t = (o:IsA("TextLabel") or o:IsA("TextButton") or o:IsA("TextBox")) and o.Text or ""
            local bn, q = parseStack(t)
            local uv = getVal(bn)
            if uv == 0 then bn, q = parseStack(o.Name); uv = getVal(bn) end
            if uv > 0 then
                tv = tv + (uv * q); ds = ds + (getDemand(bn) * q); qty = qty + q
                if uv > mv then mv = uv; mn = bn end
            end
        end
    end
    return tv, (qty > 0 and ds/qty or 1), mn, mv
end

local function fire(r)
    if not r then return false end
    return pcall(function()
        if r:IsA("RemoteEvent") then r:FireServer()
        elseif r:IsA("RemoteFunction") then r:InvokeServer() end
    end)
end

local function clickBtn(b)
    if not b or not b.Parent then return false end
    if type(firesignal) == "function" then
        pcall(firesignal, b.MouseButton1Click)
        pcall(firesignal, b.MouseButton1Down)
        pcall(firesignal, b.MouseButton1Up)
    end
    if type(getconnections) == "function" then
        local s, cs = pcall(getconnections, b.MouseButton1Click)
        if s and cs then
            for _, c in pairs(cs) do
                if c.Fire then pcall(function() c:Fire() end) end
            end
        end
    end
    return true
end

local function findBtns(root, kws)
    local out = {}
    if not root then return out end
    for _, o in ipairs(root:GetDescendants()) do
        if (o:IsA("TextButton") or o:IsA("ImageButton")) and o.Visible then
            local h = string.lower(o.Name or "")
            if o:IsA("TextButton") then h = h .. "|" .. string.lower(o.Text or "") end
            for _, k in ipairs(kws) do
                if h:find(k) and visible(o, root) then table.insert(out, o); break end
            end
        end
    end
    return out
end

local ACCEPT_KW = {"accept","aceitar","ready","pronto","confirm","confirmar","ok","yes","sim"}
local DECLINE_KW = {"decline","recusar","cancel","cancelar","reject","rejeitar","no","não","exit","sair"}

local function tryAccept(my, th)
    if my <= 0 and th <= 0 then return end
    if th <= 0 then return end
    if my > 0 and (th - my) < M.AUTO_MIN then return end
    if tick() - M.lastAutoAcceptAt < M.AUTO_CD then return end
    M.lastAutoAcceptAt = tick()
    task.spawn(function()
        fire(M.AcceptEvt)
        if M.tradeGUI then
            for _, b in ipairs(findBtns(M.tradeGUI, ACCEPT_KW)) do clickBtn(b); task.wait(0.1) end
            task.wait(0.6)
            for _, b in ipairs(findBtns(M.tradeGUI, ACCEPT_KW)) do clickBtn(b); task.wait(0.1) end
        end
    end)
end

local function tryReject(my, th)
    if not M.autoRejectLowball or my <= 0 or th <= 0 then return end
    if th >= my * 0.5 then return end
    fire(M.DeclineEvt)
    task.spawn(function()
        if M.tradeGUI then for _, b in ipairs(findBtns(M.tradeGUI, DECLINE_KW)) do clickBtn(b); task.wait(0.1) end end
    end)
end

local function mkLbl(parent, pos, size, txt, color, font, ts, align)
    local l = Instance.new("TextLabel")
    l.Size = size; l.Position = pos; l.BackgroundTransparency = 1
    l.Text = txt or ""; l.TextColor3 = color or M.C_TEXT
    l.Font = font or Enum.Font.GothamBold; l.TextSize = ts or 12
    l.TextXAlignment = align or Enum.TextXAlignment.Left

    return l
end

-- BLACK SCREEN DETECT
local function isBlackScreenActive()
    for _, gui in ipairs(PG:GetChildren()) do
        if gui:IsA("ScreenGui") and gui.Enabled and gui ~= M.gui and gui ~= M.ttGui then
            local n = string.lower(gui.Name)
            if n:find("black") or n:find("fade") or n:find("transition") or n:find("loading") or n:find("round") or n:find("cutscene") then
                for _, desc in ipairs(gui:GetDescendants()) do
                    if desc:IsA("Frame") and desc.Visible and desc.BackgroundTransparency < 0.3 then
                        local sz = desc.AbsoluteSize
                        if sz.X > 800 and sz.Y > 400 then
                            return true
                        end
                    end
                end
            end
        end
    end
    return false
end

local function hideOurGUI()
    if M.gui then M.gui.Enabled = false end
    if M.ttGui then M.ttGui.Enabled = false end
end

local function showOurGUI()
    if M.gui then M.gui.Enabled = true end
    if M.ttGui then M.ttGui.Enabled = true end
end

-- SCAN TRADE
local function scanTradeGUI()
    if M.isRespawning or not isSafe() then return end
    if isBlackScreenActive() then return end
    if not M.gui or not M.gui.Enabled then return end
    
    local found = nil
    for _, n in ipairs({"TradeGUI","TradeGui","Trade","TradingGUI","TradeMenu","TradeScreen"}) do
        local g = PG:FindFirstChild(n)
        if g and (not g:IsA("ScreenGui") or g.Enabled) then found = g; break end
    end
    if not found then
        for _, g in ipairs(PG:GetChildren()) do
            if g:IsA("ScreenGui") and g.Enabled and string.lower(g.Name):find("trade") and g ~= M.gui and g ~= M.ttGui then 
                found = g; break 
            end
        end
    end
    
    if not found then
        if M.tradeOpen then
            M.tradeOpen = false
            M.tradeGUI = nil; M.myFolder = nil; M.theirFolder = nil
            if M.gui then M.gui.DisplayOrder = 1 end
            if M.ttGui then M.ttGui.DisplayOrder = 2 end
            pcall(hideTip); pcall(resetUI)
        end
        return
    end
    
    M.tradeGUI = found
    
    local myF, theirF = nil, nil
    for _, o in ipairs(found:GetDescendants()) do
        if o:IsA("GuiObject") and o.Visible then
            local ln = string.lower(o.Name)
            if not myF and (ln:find("youroffer") or ln:find("myoffer") or ln:find("useroffer") or ln:find("playeroffer") or ln:find("youritems") or ln:find("myitems") or ln:find("playeritems")) then
                myF = o
            elseif not theirF and (ln:find("theiroffer") or ln:find("otheroffer") or ln:find("partneroffer") or ln:find("theiritems") or ln:find("otheritems") or ln:find("partneritems")) then
                theirF = o
            end
            if myF and theirF then break end
        end
    end
    
    if myF and theirF then
        M.myFolder = myF; M.theirFolder = theirF
        M.tradeOpen = true
        if M.gui then M.gui.DisplayOrder = 999 end
        if M.ttGui then M.ttGui.DisplayOrder = 1000 end
    else
        M.tradeOpen = false
    end
end

-- RESPAWN TRACKING
LP.CharacterRemoving:Connect(function()
    M.isRespawning = true
    M.tradeOpen = false
    M.tradeGUI = nil; M.myFolder = nil; M.theirFolder = nil
    pcall(hideOurGUI)
    if M.gui then M.gui.DisplayOrder = 1 end
    if M.ttGui then M.ttGui.DisplayOrder = 2 end
    pcall(hideTip); pcall(resetUI)
end)

LP.CharacterAdded:Connect(function(char)
    task.spawn(function()
        task.wait(4)
        local timeout = 0
        while timeout < 15 do
            if char and char.Parent and char:FindFirstChildOfClass("Humanoid") and char:FindFirstChildOfClass("Humanoid").Health > 0 then break end
            task.wait(0.5); timeout = timeout + 0.5
        end
        timeout = 0
        while timeout < 15 do
            if not isBlackScreenActive() then break end
            task.wait(0.5); timeout = timeout + 0.5
        end
        task.wait(2)
        M.isRespawning = false
        pcall(showOurGUI)
        if M.gui then M.gui.DisplayOrder = 1 end
        if M.ttGui then M.ttGui.DisplayOrder = 2 end
        pcall(scanTradeGUI)
    end)
end)

PG.ChildAdded:Connect(function(child)
    if child:IsA("ScreenGui") and (child.Name:find("Trade") or child.Name:find("trade")) and child ~= M.gui and child ~= M.ttGui then
        task.defer(function() task.wait(0.3); pcall(scanTradeGUI) end)
    end
end)

end)

    table.sort(list, function(a,b) return a.value > b.value end)
    local best, bestSum = nil, -1
    local function search(combo, sum, si)
        if sum <= target and sum > bestSum then
            bestSum = sum
            best = {}
            for _, it in ipairs(combo) do table.insert(best, it.name) end
        end
        if sum == target or #combo >= 4 then return end
        for i = si, #list do
            local it = list[i]
            if it.count > 0 and sum + it.value <= target then
                it.count = it.count - 1
                table.insert(combo, it)
                search(combo, sum + it.value, i)
                table.remove(combo)
                it.count = it.count + 1
            end
        end
    end
    search({}, 0, 1)
    if best and #best > 0 then
        M.lblTools.Text = string.format("🎯 Alvo: %s\nSugestão:\n%s\nTotal: %s", fmt(target), table.concat(best, " + "), fmt(bestSum))
        M.lblTools.TextColor3 = M.C_GREEN
    else
        M.lblTools.Text = "⚠ Nada encontrado"
        M.lblTools.TextColor3 = M.C_RED_BRIGHT
    end
end)

        for n, c in pairs(counts) do
            if not M.blacklist[n] and (not M.onlyDuplicates or c >= 2) then
                table.insert(list, {name=n, value=getVal(n), count=c})
            end
        end
        table.sort(list, function(x,y) return x.value > y.value end)
        local best, bestSum = nil, -1
        local function search(combo, sum, si)
            if sum <= tV and sum > bestSum then
                bestSum = sum
                best = {}
                for _, it in ipairs(combo) do table.insert(best, it.name) end
            end
            if sum == tV or #combo >= 4 then return end
            for i = si, #list do
                local it = list[i]
                if it.count > 0 and sum + it.value <= tV then
                    it.count = it.count - 1
                    table.insert(combo, it)
                    search(combo, sum + it.value, i)
                    table.remove(combo)
                    it.count = it.count + 1
                end
            end
        end
        search({}, 0, 1)
        if not best or #best == 0 then 
            M.lblAuto.Text = "⚠ Sem itens"
            M.lblAuto.TextColor3 = M.C_RED_BRIGHT
            M.isGenerating = false

            task.wait(0.2)
        end
        M.lblAuto.Text = "✔ Adicionando..."

        for _, n in ipairs(best) do
            if not M.tradeOpen then break end
            pcall(function() M.OfferEvt:FireServer(getExact(n)) end)
            task.wait(0.3)

local function getPlayerWealth(p)
    local ls = p:FindFirstChild("leaderstats")
    if ls then
        for _, n in ipairs({"Value","Coins","Money","Cash","Wealth","Gems","Wins"}) do
            local s = ls:FindFirstChild(n)
            if s and typeof(s.Value) == "number" then return s.Value end
        end
    end
    return 0
end

-- ⭐ Envia request tentando TODOS os remotes e TODOS os formatos
local function attemptSend(targetPlayer, index, total)
    if not targetPlayer or not targetPlayer.Parent then return false end
    
    local remotes = findAllSendRemotes()
    if #remotes == 0 then
        M.lblTools.Text = "⚠ NENHUM REMOTE ENCONTRADO!\nVeja o console (F9)"
        M.lblTools.TextColor3 = M.C_RED_BRIGHT
        return false
    end
    
    local succeeded = false
    local usedInfo = ""
    
    -- Tenta cada remote
    for ri, remote in ipairs(remotes) do
        if succeeded then break end
        
        -- Tenta cada formato de argumento
        local args = {
            {targetPlayer,          "Player"},
            {targetPlayer.Name,     "Name"},
            {targetPlayer.UserId,   "UserId"},
            {tostring(targetPlayer.UserId), "UserId-String"},
        }
        
        for ai, arg in ipairs(args) do
            if succeeded then break end
            local ok, err
            if remote:IsA("RemoteEvent") then
                ok, err = pcall(function() remote:FireServer(arg[1]) end)
            else
                ok, err = pcall(function() remote:InvokeServer(arg[1]) end)
            end
            
            if ok then
                succeeded = true
                usedInfo = string.format("%s / %s", remote.Name, arg[2])
                M.lastSendMethod = usedInfo
                print(string.format("[REQUEST] ✅ %s via %s → %s", targetPlayer.Name, remote.Name, arg[2]))
            end
            task.wait(0.1) -- pequeno delay entre tentativas
        end
    end
    
    if not succeeded then
        M.lblTools.Text = string.format("❌ Falha com %s\n%d remotes x 4 formatos tentados.\nVeja console (F9)", targetPlayer.Name, #remotes)
        M.lblTools.TextColor3 = M.C_RED_BRIGHT
        task.wait(2)
        return false
    end
    
    M.lblTools.Text = string.format("💎 [%d/%d] → %s\n✅ %s\nEsperando 25s...", index, total, targetPlayer.Name, usedInfo)
    M.lblTools.TextColor3 = M.C_YELLOW
    
    -- Espera 25s pela trade abrir OU player mudar
    local timer = 0
    while timer < 25 and M.isSendingRequests do
        task.wait(0.5)
        timer = timer + 0.5
        if M.tradeOpen then
            M.lblTools.Text = string.format("💎 %s ACEITOU!", targetPlayer.Name)
            M.lblTools.TextColor3 = M.C_GREEN
            M.isSendingRequests = false
            return true
        end
    end
    
    M.lblTools.Text = string.format("💎 [%d/%d] %s não respondeu.\nPróximo...", index, total, targetPlayer.Name)
    M.lblTools.TextColor3 = M.C_TEXT_DIM
    return false
end

local function runRichest()
    if M.isSendingRequests then return end
    
    -- Verifica remote antes
    local remotes = findAllSendRemotes()
    if #remotes == 0 then
        M.lblTools.Text = "⚠ NENHUM REMOTE ENCONTRADO!\nAbra o console (F9) e me mande\nos remotes que aparecem no RS."
        M.lblTools.TextColor3 = M.C_RED_BRIGHT
        return
    end
    
    M.isSendingRequests = true
    task.spawn(function()
        M.lblTools.Text = string.format("💎 Coletando TOP 5...\n(%d remotes disponíveis)", #remotes)
        M.lblTools.TextColor3 = M.C_YELLOW
        
        local list = {}
        for _, pl in ipairs(Players:GetPlayers()) do
            if pl ~= LP and pl.Character then
                table.insert(list, {player=pl, wealth=getPlayerWealth(pl), age=pl.AccountAge or 0})
            end
        end
        table.sort(list, function(a,b)
            if a.wealth ~= b.wealth then return a.wealth > b.wealth end
            return a.age > b.age
        end)
        if #list == 0 then 
            M.lblTools.Text = "💎 Nenhum jogador."
            M.lblTools.TextColor3 = M.C_TEXT_DIM
            M.isSendingRequests = false
            return 
        end
        
        local limit = math.min(#list, 5)
        for i = 1, limit do
            if not M.isSendingRequests then break end
            local pl = list[i].player
            if pl.Parent and pl.Character then
                attemptSend(pl, i, limit)
            end
        end
        if M.isSendingRequests then
            M.lblTools.Text = string.format("💎 Terminou TOP %d", limit)
            M.lblTools.TextColor3 = M.C_GREEN
            M.isSendingRequests = false
        end
    end)
end
