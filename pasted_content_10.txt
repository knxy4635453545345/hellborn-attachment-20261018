local function findRemote(cands)
    local locs = {M.TradeFolder, RS, RS:FindFirstChild("Remotes"), RS:FindFirstChild("RemoteEvents")}
    if M.TradeFolder then
        table.insert(locs, M.TradeFolder:FindFirstChild("Remotes"))
        table.insert(locs, M.TradeFolder:FindFirstChild("RemoteEvents"))
        table.insert(locs, M.TradeFolder:FindFirstChild("Events"))
    end
    for _, n in ipairs(cands) do
        for _, loc in ipairs(locs) do
            if loc then
                local r = loc:FindFirstChild(n)
                if r and (r:IsA("RemoteEvent") or r:IsA("RemoteFunction")) then
                    return r
                end
            end
        end
    end
    return nil
end

local function clean(n)
    if not n then return "" end
    n = string.lower(n)
    n = n:gsub("%b()", ""):gsub("x%d+", ""):gsub("%d+x", ""):gsub("%d+", "")
    n = n:gsub("['’]", ""):gsub("[^%w%s]", ""):gsub("%s+", " ")
    return n:match("^%s*(.-)%s*$")
end

local function parseStack(t)
    if not t then return "", 1 end
    local q = 1
    local s = t:match("^%s*(.-)%s*$") or t
    local m, qv = s:match("^(.-)%s*%(?x%s*(%d+)%)?$")
    if m and m ~= "" and qv then s = m; q = tonumber(qv) end
    local q2, m2 = s:match("^(%d+)%s*x%s*(.+)$")
    if q2 and m2 then q = tonumber(q2); s = m2 end
    return s:match("^%s*(.-)%s*$") or s, q
end

local function getVal(n)
    local c = clean(n)
    if c == "" then return 0 end
    for k, v in pairs(M.ITEM_VALUES) do if clean(k) == c then return v end end
    return 0
end

local function getDemand(n)
    local c = clean(n)
    if c == "" then return 1 end
    for k, v in pairs(M.ITEM_DEMAND) do if clean(k) == c then return v end end
    return 1
end

local function getExact(n)
    local c = clean(n)
    for k in pairs(M.ITEM_VALUES) do if clean(k) == c then return k end end
    return n
end

local function fmt(v)
    if v >= 1e9 then return string.format("%.1fB", v/1e9)
    elseif v >= 1e6 then return string.format("%.1fM", v/1e6)
    elseif v >= 1e3 then return string.format("%.1fK", v/1e3)
    else return tostring(math.floor(v)) end
end

local function visible(o, r)
    local c = o
    while c and c ~= r do
        if c:IsA("GuiObject") and not c.Visible then return false end
        c = c.Parent
    end
    return true
end

local function folderStats(f)
    if not isSafe() or not f or not f.Parent then return 0, 0 end
    local cnt, tot = 0, 0
    pcall(function()
        for _, o in ipairs(f:GetDescendants()) do
            if o:IsA("GuiObject") and o.Visible and visible(o, f) then
                local texts = {}
                if o:IsA("TextLabel") or o:IsA("TextButton") or o:IsA("TextBox") then
                    table.insert(texts, o.Text or "")
                end
                table.insert(texts, o.Name or "")
                for _, t in ipairs(texts) do
                    if t ~= "" and t ~= "..." and t ~= "???" and t ~= "Empty" and t ~= LP.Name and t ~= "Frame" and t ~= "TextLabel" and t ~= "ImageLabel" then
                        local bn, q = parseStack(t)
                        local uv = getVal(bn)
                        if uv > 0 then cnt = cnt + q; tot = tot + uv*q; break end
                    end
                end
            end
        end
    end)
    return cnt, tot
end

local function folderFull(f)
    if not isSafe() or not f or not f.Parent then return 0, 1, "—", 0 end
    local tv, qty, ds, mn, mv = 0, 0, 0, "—", 0
    pcall(function()
        for _, o in ipairs(f:GetDescendants()) do
            if o:IsA("GuiObject") and o.Visible and visible(o, f) then
                local texts = {}
                if o:IsA("TextLabel") or o:IsA("TextButton") or o:IsA("TextBox") then
                    table.insert(texts, o.Text or "")
                end
                table.insert(texts, o.Name or "")
                for _, t in ipairs(texts) do
                    if t ~= "" and t ~= "..." and t ~= "???" and t ~= "Empty" and t ~= LP.Name then
                        local bn, q = parseStack(t)
                        local uv = getVal(bn)
                        if uv > 0 then
                            tv = tv + uv*q
                            ds = ds + getDemand(bn)*q
                            qty = qty + q
                            if uv > mv then mv = uv; mn = bn end
                            break
                        end
                    end
                end
            end
        end
    end)
    return tv, (qty > 0 and ds/qty or 1), mn, mv
end

local function clickBtn(b)
    if not b or not b.Parent then return false end
    local ok = false
    if type(firesignal) == "function" then
        pcall(firesignal, b.MouseButton1Click)
        pcall(firesignal, b.MouseButton1Down)
        pcall(firesignal, b.MouseButton1Up)
        ok = true
    end
    if type(getconnections) == "function" then
        local s, cs = pcall(getconnections, b.MouseButton1Click)
        if s and cs then
            for _, c in pairs(cs) do
                if c.Fire then
                    pcall(function() c:Fire() end)
                    ok = true
                end
            end
        end
    end
    if b.AbsolutePosition and b.AbsoluteSize then
        local x = b.AbsolutePosition.X + b.AbsoluteSize.X/2
        local y = b.AbsolutePosition.Y + b.AbsoluteSize.Y/2
        pcall(function()
            VIM:SendMouseMoveEvent(x, y, game); task.wait(0.02)
            VIM:SendMouseButtonEvent(x, y, 0, true, game, 1); task.wait(0.02)
            VIM:SendMouseButtonEvent(x, y, 0, false, game, 1)
        end)
        ok = true
    end
    if type(mouse1click) == "function" then
        local x = b.AbsolutePosition.X + b.AbsoluteSize.X/2
        local y = b.AbsolutePosition.Y + b.AbsoluteSize.Y/2
        if type(mousemoveabs) == "function" then
            pcall(mousemoveabs, x, y); task.wait(0.02)
        end
        pcall(mouse1click); ok = true
    end
    return ok
end

local function findBtns(root, kws)
    local out = {}
    if not root then return out end
    for _, o in ipairs(root:GetDescendants()) do
        if (o:IsA("TextButton") or o:IsA("ImageButton")) and o.Visible then
            local h = string.lower(o.Name or "")
            if o:IsA("TextButton") then h = h .. "|" .. string.lower(o.Text or "") end
            for _, k in ipairs(kws) do
                if h:find(k) and visible(o, root) then
                    table.insert(out, o)
                    break
                end
            end
        end
    end
    return out
end

local ACCEPT_KW = {"accept","aceitar","ready","pronto","confirm","confirmar","ok","yes","sim"}
local DECLINE_KW = {"decline","recusar","cancel","cancelar","reject","rejeitar","no","não","exit","sair"}

local function tryAccept(my, th)
    if my <= 0 and th <= 0 then return end
    if th <= 0 then return end
    if my > 0 and (th - my) < M.AUTO_MIN then return end
    if tick() - M.lastAutoAcceptAt < M.AUTO_CD then return end
    M.lastAutoAcceptAt = tick()
    task.spawn(function()
        fire(M.AcceptEvt)
        local g = getTradeGUI()
        if g then
            for _, b in ipairs(findBtns(g, ACCEPT_KW)) do clickBtn(b); task.wait(0.1) end
            task.wait(0.6)
            for _, b in ipairs(findBtns(g, ACCEPT_KW)) do clickBtn(b); task.wait(0.1) end
        end
    end)
end

local function tryReject(my, th)
    if not M.autoRejectLowball then M.lowballTimer = 0; return end
    if my <= 0 or th <= 0 then M.lowballTimer = 0; return end
    if th >= my * (M.lowballThreshold/100) then M.lowballTimer = 0; return end
    if M.lowballTimer == 0 then M.lowballTimer = tick(); return end
    if tick() - M.lowballTimer < 3 then return end
    M.lowballTimer = 0
    fire(M.DeclineEvt)
    task.spawn(function()
        local g = getTradeGUI()
        if not g then return end
        for _, b in ipairs(findBtns(g, DECLINE_KW)) do clickBtn(b); task.wait(0.1) end
        task.wait(0.5)
        if g.Parent then
            for _, b in ipairs(findBtns(g, {"confirm","confirmar","yes","sim","ok"})) do
                clickBtn(b); task.wait(0.05)
            end
        end
    end)
end

local function recordPrice(name, v, side)
    if not name or v <= 0 then return end
    M.itemPriceHistory[name] = M.itemPriceHistory[name] or {}
    local h = M.itemPriceHistory[name]
    local last = h[#h]
    if last and math.abs(last.t - tick()) < 2 and last.val == v then return end
    table.insert(h, {val=v, t=tick(), side=side or "trade"})
    if #h > 50 then table.remove(h, 1) end
end

local function analyze(name)
    local h = M.itemPriceHistory[name]
    if not h or #h < 2 then return {status="no_data"} end
    local n = #h
    local cur = h[n].val
    local sma5, sma10 = nil, nil
    if n >= 5 then
        local s = 0
        for i = n-4, n do s = s + h[i].val end
        sma5 = s/5
    end
    if n >= 10 then
        local s = 0
        for i = n-9, n do s = s + h[i].val end
        sma10 = s/10
    end
    local momentum = (sma5 and sma10 and sma10 > 0) and ((sma5-sma10)/sma10*100) or 0
    local dir, arrow, dirText = "stable", "➖", "ESTÁVEL"
    if momentum > 2 then dir, arrow, dirText = "up", "📈", "SUBINDO"
    elseif momentum < -2 then dir, arrow, dirText = "down", "📉", "CAINDO" end
    local conf = math.min(100, n/8*100)
    local confLabel = conf >= 80 and "ALTA" or (conf >= 50 and "MÉDIA" or "BAIXA")
    local rec = "⏸ ESPERAR"
    if dir == "up" then rec = "🟢 COMPRAR — valor em alta"
    elseif dir == "down" then rec = "🔴 VENDER — valor em queda" end
    return {status="ok", current=cur, samples=n, momentum=momentum, arrow=arrow, dirText=dirText, confLabel=confLabel, rec=rec}
end

local function predText(name)
    local a = analyze(name)
    if a.status == "no_data" then return "📊 Sem dados (colete trades)" end
    return string.format("%s %s\n💹 Momentum: %+.1f%%\n📊 Amostras: %d (%s)\n%s", a.arrow, a.dirText, a.momentum, a.samples, a.confLabel, a.rec)
end

local function savePreds()
    if type(writefile) ~= "function" then return end
    local lines = {}
    for name, h in pairs(M.itemPriceHistory) do
        for _, e in ipairs(h) do
            table.insert(lines, string.format("%s|%d|%d|%s", name, e.val, math.floor(e.t), e.side or "trade"))
        end
    end
    pcall(writefile, "mm2_prices.dat", table.concat(lines, "\n"))
end

local function loadPreds()
    if type(readfile) ~= "function" or type(isfile) ~= "function" then return end
    if not isfile("mm2_prices.dat") then return end
    local ok, content = pcall(readfile, "mm2_prices.dat")
    if not ok or not content then return end
    for line in content:gmatch("[^\n]+") do
        local name, v, t, s = line:match("^([^|]+)|(%d+)|(%d+)|(.+)$")
        if name and v and t then
            name = name:gsub("^%s*(.-)%s*$", "%1")
            M.itemPriceHistory[name] = M.itemPriceHistory[name] or {}
            table.insert(M.itemPriceHistory[name], {val=tonumber(v), t=tonumber(t), side=s or "trade"})
        end
    end
end

local function compareItems(a, b)
    local va = getVal(a); local vb = getVal(b)
    if va == 0 then return "❌ Item A inválido" end
    if vb == 0 then return "❌ Item B inválido" end
    local lines = {}
    table.insert(lines, "┌──────────┬──────────┐")
    table.insert(lines, string.format("│ %-8s │ %-8s │", a:sub(1,8), b:sub(1,8)))
    table.insert(lines, "├──────────┼──────────┤")
    table.insert(lines, string.format("│ %-8s │ %-8s │", fmt(va), fmt(vb)))
    table.insert(lines, "└──────────┴──────────┘")
    local r = math.max(va,vb) / math.max(1, math.min(va,vb))
    if r > 1.5 then
        table.insert(lines, va > vb and string.format("🏆 %s vale %.1fx mais", a, r) or string.format("🏆 %s vale %.1fx mais", b, r))
    else
        table.insert(lines, "⚖ Valores próximos")
    end
    return table.concat(lines, "\n")
end

        local m = UIS:GetMouseLocation()
        local vp = Vector2.new(1920,1080)
        if workspace.CurrentCamera then vp = workspace.CurrentCamera.ViewportSize end
        M.ttFrame.Position = UDim2.fromOffset(math.max(0, math.min(m.X + 15, vp.X - 300)), math.max(0, math.min(m.Y + 15, vp.Y - 180)))

local function hideTip()
    M.ttFrame.Visible = false
    if M.ttConn then M.ttConn:Disconnect(); M.ttConn = nil end
end

local function attachTips(folder, side)
    if not folder or not folder.Parent then return end
    for _, o in ipairs(folder:GetDescendants()) do
        if o:IsA("TextLabel") and o.Visible and not M.attached[o] then
            local bn = parseStack(o.Text)
            if getVal(bn) > 0 then
                M.attached[o] = true
                o.MouseEnter:Connect(function() showTip(bn, side) end)
                o.MouseLeave:Connect(hideTip)
                local p = o.Parent
                if p and p:IsA("GuiObject") and not M.attached[p] then
                    M.attached[p] = true
                    p.MouseEnter:Connect(function() showTip(bn, side) end)
                    p.MouseLeave:Connect(hideTip)
                end
            end
        end
    end
end

local function updateStats()
    local t = string.format("📊 %d trades | Lucro: %s", M.stats.trades, fmt(M.stats.profit))
    if M.stats.wins > 0 or M.stats.losses > 0 then
        t = t .. string.format(" (%d✔/%d✖)", M.stats.wins, M.stats.losses)
    end
    M.lblStats.Text = t
    if M.stats.profit > 0 then M.lblStats.TextColor3 = M.C_GREEN
    elseif M.stats.profit < 0 then M.lblStats.TextColor3 = M.C_RED_BRIGHT
    else M.lblStats.TextColor3 = M.C_TEXT_DIM end
end

    for k in pairs(M.blacklist) do table.insert(items, k) end

local function updPersonal()
    if M.personalStats.tradesCount == 0 then
        M.lblPersonal.Text = "📊 Sem trades ainda"
        return
    end
    local avg = M.personalStats.profitSum / M.personalStats.tradesCount
    local wr = M.stats.trades > 0 and (M.stats.wins/M.stats.trades*100) or 0
    local tops = {}
    for k, c in pairs(M.personalStats.itemsTraded) do table.insert(tops, {n=k, c=c}) end
    table.sort(tops, function(a,b) return a.c > b.c end)
    local top3 = {}
    for i = 1, math.min(3, #tops) do table.insert(top3, string.format("%s (%dx)", tops[i].n, tops[i].c)) end
    M.lblPersonal.Text = string.format("🏆 Melhor: %s\n💀 Pior: %s\n📊 Média: %s\n✅ %d✔ / %d✖ (%.0f%%)\n🔥 Top: %s",
        fmt(M.personalStats.best.profit), fmt(M.personalStats.worst.profit), fmt(avg),
        M.stats.wins, M.stats.losses, wr,
        #top3 > 0 and table.concat(top3, ", ") or "—")
end

local function updFlip()
    if M.stats.trades == 0 then M.lblFlip.Text = "⏳ Aguardando..."; return end
    local el = math.max(1, tick() - M.flip.time)
    local tD = M.stats.trades - M.flip.count
    local pD = M.stats.profit - M.flip.profit
    if tD == 0 then M.lblFlip.Text = "⏳ Aguardando..."; return end
    local avg = pD / tD
    local tpm = tD / (el / 60)
    local hr = avg * tpm * 60
    M.lblFlip.Text = string.format("📈 Média: %s/trade\n⚡ %.1f trades/min\n💰 Lucro/h: %s\n📅 Dia: %s\n🏆 Semana: %s",
        fmt(avg), tpm, fmt(hr), fmt(hr*4), fmt(hr*4*7))
end

            M.lastTheirVal = 0
        end
    end)

M.btnRev.MouseButton1Click:Connect(function()
    local target = tonumber(M.inTarget.Text)
    if not target or target <= 0 then M.lblTools.Text = "⚠ Valor inválido"; return end
    local inv = getInv()
    local counts = {}
    for _, n in ipairs(inv) do counts[n] = (counts[n] or 0) + 1 end
    local list = {}
    for n, c in pairs(counts) do
        if not M.blacklist[n] then
            table.insert(list, {name=n, value=getVal(n), count=c})
        end
    end
    table.sort(list, function(a,b) return a.value > b.value end)
    local best, bestSum = nil, -1
    local function search(combo, sum, si)
        if sum <= target and sum > bestSum then
            bestSum = sum
            best = {}
            for _, it in ipairs(combo) do table.insert(best, it.name) end
        end
        if sum == target or #combo >= 4 then return end
        for i = si, #list do
            local it = list[i]
            if it.count > 0 and sum + it.value <= target then
                it.count = it.count - 1
                table.insert(combo, it)
                search(combo, sum + it.value, i)
                table.remove(combo)
                it.count = it.count + 1
            end
        end
    end

-- Generate
local function genOffer()
    if M.isGenerating or not isSafe() then return end
    local o, _, tf = isOpen()
    if not o then M.lblAuto.Text = "⚠ Abra trade"; return end
    if not M.OfferEvt then M.lblAuto.Text = "⚠ Remote não achado"; return end
    local _, tV = folderStats(tf)
    if tV <= 0 then M.lblAuto.Text = "⚠ Parceiro sem itens"; return end
    M.isGenerating = true
    M.lblAuto.Text = "🔍 Calculando..."
    task.spawn(function()
        local inv = getInv()
        local counts = {}
        for _, n in ipairs(inv) do counts[n] = (counts[n] or 0) + 1 end
        local list = {}
        for n, c in pairs(counts) do
            if not M.blacklist[n] and (not M.onlyDuplicates or c >= 2) then
                table.insert(list, {name=n, value=getVal(n), count=c})
            end
        end
        table.sort(list, function(a,b) return a.value > b.value end)
        local best, bestSum = nil, -1
        local function search(combo, sum, si)
            if sum <= tV and sum > bestSum then
                bestSum = sum
                best = {}
                for _, it in ipairs(combo) do table.insert(best, it.name) end
            end
            if sum == tV or #combo >= 4 then return end
            for i = si, #list do
                local it = list[i]
                if it.count > 0 and sum + it.value <= tV then
                    it.count = it.count - 1
                    table.insert(combo, it)
                    search(combo, sum + it.value, i)
                    table.remove(combo)
                    it.count = it.count + 1
                end
            end
        end
        search({}, 0, 1)
        if not best or #best == 0 then
            M.lblAuto.Text = "⚠ Sem itens"
            M.isGenerating = false
            return
        end
        M.lblAuto.Text = "🧹 Limpando..."
        local safety = 0
        while M.isGenerating and isSafe() do
            local oo, mf = isOpen()
            if not oo then M.isGenerating = false; return end
            if folderStats(mf) <= 0 or safety >= 20 then break end
            safety = safety + 1
            local f = nil
            for _, obj in ipairs(mf:GetDescendants()) do
                if obj:IsA("GuiObject") and obj.Visible then
                    local t = (obj:IsA("TextLabel") and obj.Text) or obj.Name or ""
                    local bn = parseStack(t)
                    if getVal(bn) > 0 then f = bn; break end
                end
            end
            if not f then break end
            if M.RemoveEvt then pcall(function() M.RemoveEvt:FireServer(getExact(f)) end) end
            task.wait(0.2)
        end
        M.lblAuto.Text = "✔ Adicionando..."
        local fs = 0
        for _, n in ipairs(best) do
            local oo, mf = isOpen()
            if not oo then M.isGenerating = false; return end
            local before = folderStats(mf)
            pcall(function() M.OfferEvt:FireServer(getExact(n)) end)
            for _ = 1, 8 do
                task.wait(0.15)
                if folderStats(mf) > before then fs = fs + getVal(n); break end
            end
        end
        M.lblAuto.Text = string.format("✔ Seu: %s | Deles: %s\nLucro: %s", fmt(fs), fmt(tV), fmt(tV - fs))
        M.lblAuto.TextColor3 = M.C_GREEN
        M.isGenerating = false
    end)
end

-- Tools Functions
local function getPlayerWealth(p)
    local ls = p:FindFirstChild("leaderstats")
    if ls then
        for _, n in ipairs({"Value","Coins","Money","Cash","Wealth","Gems","Wins"}) do
            local s = ls:FindFirstChild(n)
            if s and typeof(s.Value) == "number" then return s.Value, "leaderstats:"..n end
        end
    end
    local st = p:FindFirstChild("Stats") or p:FindFirstChild("PlayerStats")
    if st then
        for _, n in ipairs({"Value","Coins","Money","Cash"}) do
            local s = st:FindFirstChild(n)
            if s and typeof(s.Value) == "number" then return s.Value, "stats:"..n end
        end
    end
    return 0, "unknown"
end

local function sendAndWait(targetPlayer, index, total)
    if not targetPlayer or not targetPlayer.Parent then return false end
    
    local remote = getSendTradeRemote()
    if not remote then
        M.lblTools.Text = "⚠ Remote SendTradeRequest NÃO encontrado no jogo."
        M.lblTools.TextColor3 = M.C_RED_BRIGHT
        return false
    end
    
    local success, err = false, ""
    if remote:IsA("RemoteEvent") then
        success, err = pcall(function() remote:FireServer(targetPlayer) end)
        if not success then
            success, err = pcall(function() remote:FireServer(targetPlayer.Name) end)
        end
    else
        success, err = pcall(function() remote:InvokeServer(targetPlayer) end)
        if not success then
            success, err = pcall(function() remote:InvokeServer(targetPlayer.Name) end)
        end
    end
    
    if not success then
        M.lblTools.Text = string.format("❌ Erro ao enviar para %s:\n%s\nTentando próximo...", targetPlayer.Name, tostring(err))
        M.lblTools.TextColor3 = M.C_RED_BRIGHT
        task.wait(2)
        return false
    end
    
    -- Espera 25 segundos
    local waitTime = 25
    local timer = 0
    local accepted = false
    
    while timer < waitTime and M.isSendingRequests do
        task.wait(0.5)
        timer = timer + 0.5
        
        if isOpen() then
            accepted = true
            break
        end
    end
    
    if accepted then
        M.lblTools.Text = string.format("💎 %s ACEITOU!\nPausando envio para negociar.", targetPlayer.Name)
        M.lblTools.TextColor3 = M.C_GREEN
        M.isSendingRequests = false
    else
        M.lblTools.Text = string.format("💎 [%d/%d] %s ignorou/recusou.\nIndo para o próximo...", index, total, targetPlayer.Name)
        M.lblTools.TextColor3 = M.C_TEXT_DIM
    end
    return accepted
end

                            M.lastAutoAcceptAt = 0
                            pcall(tryAccept, mV, tV)
                        end
                    end
                end
            else
                if wasOpen then
                    wasOpen = false
                    M.currentTradeSeen = {}
                    hideTip()
                    resetUI()
                end
            end
        end
    end
end)
