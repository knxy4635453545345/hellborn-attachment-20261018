local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local LocalPlayer = Players.LocalPlayer

-- ================== CONFIGURAÇÕES ==================
local CONFIG = {
    TweenSpeed = 12,          -- Studs por segundo (recomendado: 10-15)
    MaxCoins = 50,            -- ✅ LIMITE DE 50 MOEDAS
    CheckInterval = 1.5,
    AutoFarmEnabled = false,  -- Começa desligado; ligue pela GUI
}

-- ================== ESTADO ==================
local currentTween = nil
local coinsCollected = 0
local isRunning = false

-- ================== FUNÇÕES AUXILIARES ==================
local function getCoinCount()
    local coins = LocalPlayer:FindFirstChild("Coins") or LocalPlayer:FindFirstChild("coins")
    if coins and typeof(coins) == "ValueBase" then
        return coins.Value
    end
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        local count = 0
        for _, item in ipairs(backpack:GetChildren()) do
            if item.Name:lower():find("coin") then
                count = count + 1
            end
        end
        return count
    end
    return 0
end

local function isInventoryFull()
    return getCoinCount() >= CONFIG.MaxCoins
end

local function flyTo(position, speed)
    if not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        return
    end
    local hrp = LocalPlayer.Character.HumanoidRootPart
    local distance = (position - hrp.Position).Magnitude
    local duration = distance / speed
    if currentTween then
        currentTween:Cancel()
    end
    local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear)
    currentTween = TweenService:Create(hrp, tweenInfo, { CFrame = CFrame.new(position) })
    currentTween:Play()
    currentTween.Completed:Wait()
end

local function getNearestCoin()
    local nearest = nil
    local shortestDist = math.huge
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name:lower():find("coin") then
            local dist = (obj.Position - hrp.Position).Magnitude
            if dist < shortestDist then
                shortestDist = dist
                nearest = obj
            end
        end
    end
    return nearest
end

local function getRole()
    local roleObj = LocalPlayer:FindFirstChild("Role") or workspace:FindFirstChild(LocalPlayer.Name .. "_Role")
    if roleObj and roleObj:IsA("StringValue") then
        return roleObj.Value:lower()
    end
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, tool in ipairs(backpack:GetChildren()) do
            if tool.Name:lower():find("knife") then return "murderer" end
            if tool.Name:lower():find("gun") then return "sheriff" end
        end
    end
    return "innocent"
end

-- ================== AÇÃO POR PAPEL ==================
local function executeRoleAction()
    local role = getRole()

    if role == "sheriff" and isInventoryFull() then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                local char = player.Character
                if char then
                    local isMurderer = false
                    local bp = char:FindFirstChild("Backpack")
                    if bp then
                        for _, tool in ipairs(bp:GetChildren()) do
                            if tool.Name:lower():find("knife") then isMurderer = true end
                        end
                    end
                    if isMurderer then
                        local targetHrp = char:FindFirstChild("HumanoidRootPart")
                        if targetHrp then
                            flyTo(targetHrp.Position, CONFIG.TweenSpeed)
                            local gun = LocalPlayer.Backpack:FindFirstChildOfClass("Tool")
                            if gun then gun:Activate() end
                        end
                        break
                    end
                end
            end
        end

    elseif role == "murderer" and isInventoryFull() then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local targetHrp = player.Character:FindFirstChild("HumanoidRootPart")
                if targetHrp then
                    flyTo(targetHrp.Position, CONFIG.TweenSpeed)
                    local knife = LocalPlayer.Backpack:FindFirstChildOfClass("Tool")
                    if knife then knife:Activate() end
                    task.wait(0.5)
                end
            end
        end
    end
end

-- ================== LOOP PRINCIPAL ==================
local function startFarmLoop()
    if isRunning then return end
    isRunning = true
    task.spawn(function()
        while CONFIG.AutoFarmEnabled do
            if not isInventoryFull() then
                local coin = getNearestCoin()
                if coin then
                    flyTo(coin.Position, CONFIG.TweenSpeed)
                    coinsCollected = coinsCollected + 1
                    task.wait(0.3)
                else
                    task.wait(1)
                end
            else
                executeRoleAction()
                task.wait(CONFIG.CheckInterval)
            end
            RunService.Heartbeat:Wait()
        end
        isRunning = false
    end)
end

-- ================== GUI ==================
